CREATE OR REPLACE FUNCTION private.enforce_campaign_owner()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
begin
  if not exists (
    select 1
    from public.project_members member
    where member.project_id = new.project_id
      and member.profile_id = new.owner_id
      and member.organization_id = new.organization_id
  ) then
    raise exception 'campaign owner must be a member of the project';
  end if;
  return new;
end;
$function$

CREATE OR REPLACE FUNCTION private.enforce_member_flags()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
begin
  if new.role = 'viewer' then
    new.can_edit = false;
    new.can_manage_members = false;
    new.can_approve_deliverables = false;
    new.can_manage_assets = false;
    new.can_comment = false;
  end if;

  if current_user in ('authenticated', 'anon') and new.can_approve_deliverables then
    if tg_op = 'INSERT' then
      if not private.is_org_admin(new.organization_id) then
        raise exception 'only an organization admin can grant deliverable approval'
          using errcode = '42501';
      end if;
    elsif new.can_approve_deliverables is distinct from old.can_approve_deliverables
      and not private.is_org_admin(new.organization_id) then
      raise exception 'only an organization admin can grant deliverable approval'
        using errcode = '42501';
    end if;
  end if;

  return new;
end;
$function$

CREATE OR REPLACE FUNCTION private.enforce_signed_in_owner()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
begin
  if auth.uid() is not null and new.owner_id is distinct from auth.uid() then
    raise exception 'owner_id must be the signed-in user';
  end if;
  return new;
end;
$function$

CREATE OR REPLACE FUNCTION private.enforce_task_rules()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
begin
  if new.campaign_id is not null and not exists (
    select 1
    from public.campaigns campaign
    where campaign.id = new.campaign_id
      and campaign.project_id = new.project_id
      and campaign.organization_id = new.organization_id
  ) then
    raise exception 'task campaign must belong to the same project';
  end if;

  if new.assignee_id is not null and not exists (
    select 1
    from public.project_members member
    where member.project_id = new.project_id
      and member.profile_id = new.assignee_id
      and member.organization_id = new.organization_id
  ) then
    raise exception 'assignee must be a member of the project';
  end if;

  return new;
end;
$function$

CREATE OR REPLACE FUNCTION private.ensure_project_owner()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
begin
  insert into public.project_members (
    organization_id, project_id, profile_id, role,
    can_edit, can_manage_members, can_approve_deliverables, can_manage_assets, can_comment
  )
  values (
    new.organization_id, new.id, new.owner_id, 'project_manager',
    true, true, false, true, true
  )
  on conflict (project_id, profile_id) do nothing;

  if auth.uid() is not null and auth.uid() is distinct from new.owner_id then
    insert into public.project_members (
      organization_id, project_id, profile_id, role,
      can_edit, can_manage_members, can_approve_deliverables, can_manage_assets, can_comment
    )
    values (
      new.organization_id, new.id, auth.uid(), 'project_manager',
      true, true, false, true, true
    )
    on conflict (project_id, profile_id) do nothing;
  end if;

  insert into public.project_departments (organization_id, project_id, department_id)
  values (new.organization_id, new.id, new.lead_department_id)
  on conflict (project_id, department_id) do nothing;

  return null;
end;
$function$

CREATE OR REPLACE FUNCTION private.preserve_last_org_admin()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
begin
  if TG_OP = 'DELETE' and OLD.role = 'org_admin' then
    perform private.sec01_lock_organization(OLD.organization_id);
    if private.sec01_organization_exists(OLD.organization_id) and not exists (
      select 1 from public.organization_memberships m
      where m.organization_id = OLD.organization_id
        and m.role = 'org_admin'
        and m.id <> OLD.id
    ) then
      raise exception 'Cannot delete the last org_admin from an organization';
    end if;
    return OLD;
  end if;
  if TG_OP = 'UPDATE' and OLD.role = 'org_admin' and (NEW.role <> 'org_admin' or NEW.organization_id is distinct from OLD.organization_id) then
    perform private.sec01_lock_organization(OLD.organization_id);
    if not exists (
      select 1 from public.organization_memberships m
      where m.organization_id = OLD.organization_id
        and m.role = 'org_admin'
        and m.id <> OLD.id
    ) then
      raise exception 'Cannot delete the last org_admin from an organization';
    end if;
  end if;
  if TG_OP = 'DELETE' then
    return OLD;
  end if;
  return NEW;
end;
$function$

CREATE OR REPLACE FUNCTION private.protect_project_owner_membership()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
begin
  if exists (
    select 1
    from public.projects project
    where project.id = old.project_id
      and project.owner_id = old.profile_id
  ) then
    raise exception 'reassign the project owner before removing that membership';
  end if;
  return old;
end;
$function$
